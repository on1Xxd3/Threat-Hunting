###Приведение типов и “аккуратные данные”
# ---- вспомогательные детекторы типов ----

looks_like_datetime <- function(col) {
if (is.character(col) || is.factor(col)) {
sample_vals <- as.character(col)
sample_vals <- sample_vals[!is.na(sample_vals)]
if (length(sample_vals) == 0) return(FALSE)
probe <- try(as.POSIXct(sample_vals[1]), silent = TRUE)
return(inherits(probe, "POSIXct"))
}
FALSE
}

looks_like_number <- function(col) {
if (is.character(col) || is.factor(col)) {
x <- as.character(col)
x <- x[!is.na(x)]
suppressWarnings(all(!is.na(as.numeric(x))))
} else {
FALSE
}
}

looks_like_logical <- function(col) {
if (is.character(col) || is.factor(col)) {
x <- tolower(as.character(col))
uniq <- unique(x[!is.na(x)])
allowed <- c("true", "false", "1", "0")
ok <- length(uniq) > 0 && all(uniq %in% allowed)
has_tf <- any(uniq == "true") && any(uniq == "false")
has_10 <- any(uniq == "1") && any(uniq == "0")
ok && (has_tf || has_10)
} else {
FALSE
}
}

# ---- datetime ----

dt_cols <- names(log_raw)[sapply(log_raw, looks_like_datetime)]
if (length(dt_cols) > 0) {
for (nm in dt_cols) {
log_raw[[nm]] <- tryCatch(as.POSIXct(log_raw[[nm]]), error = function(e) log_raw[[nm]])
}
}

# ---- numeric ----

num_cols <- names(log_raw)[sapply(log_raw, looks_like_number)]
if (length(num_cols) > 0) {
log_raw <- log_raw %>% mutate(across(all_of(num_cols), ~as.numeric(as.character(.x))))
}

# ---- logical ----

bool_cols <- names(log_raw)[sapply(log_raw, looks_like_logical)]
if (length(bool_cols) > 0) {
for (nm in bool_cols) {
v <- tolower(as.character(log_raw[[nm]]))
log_raw[[nm]] <- ifelse(v %in% c("true", "1"), TRUE,
ifelse(v %in% c("false", "0"), FALSE, NA))
}
}
